cmake_minimum_required(VERSION 3.16)
project(SuperJupiterIntegrationConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_INTEGRATION_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_INTEGRATION_SOURCE_DIR}" integration)
  if(TARGET superjupiter-integration-tests OR TARGET superjupiter-integration-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterIntegration 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-integration-consumer main.cpp)
target_link_libraries(superjupiter-integration-consumer PRIVATE SuperJupiter::Integration)
